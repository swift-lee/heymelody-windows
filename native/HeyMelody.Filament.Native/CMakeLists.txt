cmake_minimum_required(VERSION 3.22.1)
if(CMAKE_HOST_APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 15.0 CACHE STRING "Native bridge minimum macOS version")
endif()
project(HeyMelodyFilamentNative VERSION 2.0 LANGUAGES C CXX)

set(HM_RUNTIME_IDENTIFIER "" CACHE STRING "Exact runtime architecture being built")
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" HM_TARGET_PROCESSOR)
if(NOT HM_RUNTIME_IDENTIFIER)
    if(WIN32)
        set(HM_RUNTIME_IDENTIFIER "win-x64")
    elseif(APPLE)
        if(CMAKE_OSX_ARCHITECTURES STREQUAL "x86_64" OR (NOT CMAKE_OSX_ARCHITECTURES AND HM_TARGET_PROCESSOR MATCHES "x86_64|amd64"))
            set(HM_RUNTIME_IDENTIFIER "osx-x64")
        else()
            set(HM_RUNTIME_IDENTIFIER "osx-arm64")
        endif()
    elseif(HM_TARGET_PROCESSOR MATCHES "aarch64|arm64")
        set(HM_RUNTIME_IDENTIFIER "linux-arm64")
    else()
        set(HM_RUNTIME_IDENTIFIER "linux-x64")
    endif()
endif()
if(HM_RUNTIME_IDENTIFIER STREQUAL "win-x64")
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT HM_TARGET_PROCESSOR MATCHES "^(amd64|x86_64|x64)$")
        message(FATAL_ERROR "win-x64 requires an actual Windows x64 compiler target")
    endif()
elseif(HM_RUNTIME_IDENTIFIER STREQUAL "win-arm64")
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT HM_TARGET_PROCESSOR MATCHES "^(aarch64|arm64)$" OR
       (MSVC AND NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "ARM64"))
        message(FATAL_ERROR "win-arm64 requires an actual Windows ARM64 compiler target")
    endif()
elseif(HM_RUNTIME_IDENTIFIER MATCHES "^osx-(x64|arm64)$")
    if(NOT APPLE)
        message(FATAL_ERROR "macOS runtime requires a macOS toolchain")
    endif()
    set(HM_EXPECTED_OSX_ARCH arm64)
    if(HM_RUNTIME_IDENTIFIER STREQUAL "osx-x64")
        set(HM_EXPECTED_OSX_ARCH x86_64)
    endif()
    if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL HM_EXPECTED_OSX_ARCH)
        message(FATAL_ERROR "Set CMAKE_OSX_ARCHITECTURES=${HM_EXPECTED_OSX_ARCH} for ${HM_RUNTIME_IDENTIFIER}")
    endif()
elseif(HM_RUNTIME_IDENTIFIER MATCHES "^linux-(x64|arm64)$")
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "Linux runtime requires a Linux toolchain")
    endif()
    if((HM_RUNTIME_IDENTIFIER STREQUAL "linux-x64" AND NOT HM_TARGET_PROCESSOR MATCHES "^(amd64|x86_64|x64)$") OR
       (HM_RUNTIME_IDENTIFIER STREQUAL "linux-arm64" AND NOT HM_TARGET_PROCESSOR MATCHES "^(aarch64|arm64)$"))
        message(FATAL_ERROR "${HM_RUNTIME_IDENTIFIER} does not match compiler processor ${CMAKE_SYSTEM_PROCESSOR}; provide a matching host/toolchain")
    endif()
else()
    message(FATAL_ERROR "Unsupported runtime identifier: ${HM_RUNTIME_IDENTIFIER}")
endif()

set(HM_FILAMENT_COMMIT "ea31afeaf66339e7414aa648aa76c7e61adbde28")
set(HM_FILAMENT_SOURCE "" CACHE PATH "Explicit pinned Filament source directory")
if(NOT EXISTS "${HM_FILAMENT_SOURCE}/filament/include/filament/Engine.h")
    message(FATAL_ERROR "Set HM_FILAMENT_SOURCE to the pinned Filament checkout; research is never implicitly used.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(FILAMENT_SKIP_SAMPLES ON CACHE BOOL "" FORCE)
set(FILAMENT_SKIP_SDL2 ON CACHE BOOL "" FORCE)
set(FILAMENT_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(FILAMENT_BUILD_JAVA OFF CACHE BOOL "" FORCE)
set(FILAMENT_ENABLE_MATDBG OFF CACHE BOOL "" FORCE)
set(FILAMENT_ENABLE_FGVIEWER OFF CACHE BOOL "" FORCE)
set(FILAMENT_SUPPORTS_WEBGPU OFF CACHE BOOL "" FORCE)
if(APPLE)
    enable_language(OBJCXX)
    set(FILAMENT_SUPPORTS_VULKAN OFF CACHE BOOL "" FORCE)
    set(FILAMENT_SUPPORTS_METAL ON CACHE BOOL "" FORCE)
    set(FILAMENT_SUPPORTS_OPENGL OFF CACHE BOOL "" FORCE)
else()
    set(FILAMENT_SUPPORTS_VULKAN ON CACHE BOOL "" FORCE)
    set(FILAMENT_SUPPORTS_METAL OFF CACHE BOOL "" FORCE)
    set(FILAMENT_SUPPORTS_OPENGL ON CACHE BOOL "" FORCE)
    if(UNIX)
        # Keep the parent bridge and Filament on Clang's default dynamic libstdc++.
        set(USE_STATIC_LIBCXX OFF CACHE BOOL "Use the same C++ runtime as the native bridge" FORCE)
        set(FILAMENT_SUPPORTS_EGL_ON_LINUX ON CACHE BOOL "" FORCE)
        set(FILAMENT_SUPPORTS_XCB OFF CACHE BOOL "" FORCE)
        set(FILAMENT_SUPPORTS_XLIB OFF CACHE BOOL "" FORCE)
    endif()
endif()

# This helper intentionally has a native dependency on the sibling bridge. Windows
# resolves it beside the executable; Unix embeds a relative runtime search path.
add_executable(hm_filament_probe tools/probe.cpp)
target_link_libraries(hm_filament_probe PRIVATE HeyMelody.Filament.Native)
target_include_directories(hm_filament_probe PRIVATE include)
if(MSVC)
    set_property(TARGET hm_filament_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
elseif(APPLE)
    set_target_properties(hm_filament_probe PROPERTIES BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "@loader_path")
else()
    set_target_properties(hm_filament_probe PROPERTIES BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH "$ORIGIN")
endif()
if(DEFINED HM_RUNTIME_OUTPUT_DIRECTORY)
    set_target_properties(hm_filament_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${HM_RUNTIME_OUTPUT_DIRECTORY}")
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    include(CheckCXXSymbolExists)
    # Recheck the actual standard library when compiler flags change on reconfigure.
    unset(HM_FILAMENT_USES_LIBSTDCXX CACHE)
    check_cxx_symbol_exists(__GLIBCXX__ "cstddef" HM_FILAMENT_USES_LIBSTDCXX)
    if(HM_FILAMENT_USES_LIBSTDCXX)
        # Upstream defines libc++ thread annotations even when using libstdc++.
        # Directory options follow its CMAKE_CXX_FLAGS -D, so this -U takes effect.
        # Keep annotations enabled for builds that actually use libc++.
        add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-U_LIBCPP_ENABLE_THREAD_SAFETY_ANNOTATIONS>")
    endif()
endif()

add_subdirectory("${HM_FILAMENT_SOURCE}" filament EXCLUDE_FROM_ALL)
if(APPLE)
    # The pinned backend still maps PVRTC enum values deprecated by the macOS 15
    # SDK. Keep its source intact while permitting those compatibility mappings
    # with current Xcode; other warnings remain errors in the upstream target.
    target_compile_options(backend PRIVATE -Wno-error=deprecated-declarations)
endif()
if(HM_RUNTIME_IDENTIFIER STREQUAL "win-arm64")
    # Upstream selects x64 MASM for every 64-bit Windows target. Its existing typed
    # C++ dispatch wrappers preserve the target ABI and are also valid on ARM64.
    # Change only this target; the pinned upstream checkout remains unmodified.
    get_target_property(HM_BLUEGL_SOURCES bluegl SOURCES)
    list(FILTER HM_BLUEGL_SOURCES EXCLUDE REGEX "BlueGLCoreWindowsImpl\\.S$")
    set_property(TARGET bluegl PROPERTY SOURCES "${HM_BLUEGL_SOURCES}")
    target_sources(bluegl PRIVATE "${HM_FILAMENT_SOURCE}/libs/bluegl/src/BlueGLCoreWindows32Impl.cpp")
endif()

add_library(HeyMelody.Filament.Native SHARED src/hm_filament.cpp src/hm_sha256.cpp)
target_compile_definitions(HeyMelody.Filament.Native PRIVATE HM_FILAMENT_BUILD=1)
target_include_directories(HeyMelody.Filament.Native PUBLIC include PRIVATE
    "${HM_FILAMENT_SOURCE}/third_party/cgltf"
    "${CMAKE_CURRENT_BINARY_DIR}/filament/libs/gltfio")
target_link_libraries(HeyMelody.Filament.Native PRIVATE filament gltfio_core uberarchive ktxreader image)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # The EGL backend calls GLES directly. Attach its dependency to the static
    # archive so consumers link GLES after it, including with --as-needed.
    find_library(HM_GLESV2_LIBRARY NAMES GLESv2 REQUIRED)
    target_link_libraries(backend PRIVATE "${HM_GLESV2_LIBRARY}")
endif()
set_target_properties(HeyMelody.Filament.Native PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    WINDOWS_EXPORT_ALL_SYMBOLS OFF)
if(MSVC)
    set_property(TARGET HeyMelody.Filament.Native PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_compile_options(HeyMelody.Filament.Native PRIVATE /W4 /EHsc)
else()
    target_compile_options(HeyMelody.Filament.Native PRIVATE -Wall -Wextra)
endif()
if(WIN32)
    target_link_libraries(HeyMelody.Filament.Native PRIVATE bcrypt)
endif()
if(DEFINED HM_RUNTIME_OUTPUT_DIRECTORY)
    set_target_properties(HeyMelody.Filament.Native PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${HM_RUNTIME_OUTPUT_DIRECTORY}"
        LIBRARY_OUTPUT_DIRECTORY "${HM_RUNTIME_OUTPUT_DIRECTORY}")
endif()
