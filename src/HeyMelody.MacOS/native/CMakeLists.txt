cmake_minimum_required(VERSION 3.20)
project(HeyMelodyMac LANGUAGES CXX OBJCXX)
if(NOT APPLE)
    message(FATAL_ERROR "HeyMelodyMac requires macOS")
endif()
set(CMAKE_OSX_ARCHITECTURES arm64)
set(CMAKE_OSX_DEPLOYMENT_TARGET 15.0)
set(CMAKE_CXX_STANDARD 17)
file(GLOB BRIDGE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.mm")
add_library(HeyMelodyMac SHARED ${BRIDGE_SOURCES})
target_compile_options(HeyMelodyMac PRIVATE -fobjc-arc -fblocks)
target_link_libraries(HeyMelodyMac PRIVATE "-framework Foundation" "-framework AppKit" "-framework IOBluetooth" "-framework CoreBluetooth" "-framework CoreAudio" "-framework AudioToolbox")
set_target_properties(HeyMelodyMac PROPERTIES INSTALL_NAME_DIR "@rpath")
